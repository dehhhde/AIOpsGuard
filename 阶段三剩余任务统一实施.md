"""数据库连接管理"""
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker
from sqlalchemy.orm import declarative_base
from contextlib import asynccontextmanager
from typing import Optional
import structlog

from config.settings import settings

logger = structlog.get_logger(__name__)

# 声明基类
Base = declarative_base()

# 全局引擎和会话工厂
_engine: Optional[any] = None
_session_factory: Optional[any] = None


def init_engine():
    """初始化数据库引擎"""
    global _engine, _session_factory
    
    if not settings.database_enabled:
        logger.info("database_disabled_skip_init")
        return
    
    try:
        _engine = create_async_engine(
            settings.database_url,
            echo=settings.database_echo,
            pool_size=settings.database_pool_size,
            max_overflow=settings.database_max_overflow,
            pool_pre_ping=True
        )
        
        _session_factory = async_sessionmaker(
            _engine,
            class_=AsyncSession,
            expire_on_commit=False
        )
        
        logger.info("database_engine_initialized", url=settings.database_url)
    except Exception as e:
        logger.error("database_init_failed", error=str(e))
        _engine = None
        _session_factory = None


@asynccontextmanager
async def get_db_session():
    """获取数据库会话
    
    使用上下文管理器自动管理事务
    """
    if not _session_factory:
        logger.warning("database_not_initialized_skip_session")
        yield None
        return
    
    async with _session_factory() as session:
        try:
            yield session
            await session.commit()
        except Exception as e:
            await session.rollback()
            logger.error("database_transaction_failed", error=str(e))
            raise
        finally:
            await session.close()


async def init_db():
    """初始化数据库（创建表）"""
    if not _engine:
        logger.warning("database_not_initialized_skip_create_tables")
        return
    
    try:
        async with _engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
        logger.info("database_tables_created")
    except Exception as e:
        logger.error("database_create_tables_failed", error=str(e))


async def close_db():
    """关闭数据库连接"""
    if _engine:
        await _engine.dispose()
        logger.info("database_connection_closed")
```

"""诊断会话映射器"""
from domain.models.diagnosis_session import DiagnosisSession
from domain.value_objects.time_range import TimeRange
from domain.value_objects.service_identifier import ServiceIdentifier
from ..models.diagnosis_orm import DiagnosisSessionORM


class DiagnosisMapper:
    """领域模型 ↔ ORM 映射器"""
    
    @staticmethod
    def to_orm(session: DiagnosisSession) -> DiagnosisSessionORM:
        """领域模型转 ORM"""
        return DiagnosisSessionORM(
            id=session.id,
            session_id=session.session_id,
            service_name=session.service_identifier.service_name,
            service_instance=session.service_identifier.service_instance,
            environment=session.service_identifier.environment,
            start_time=session.time_range.start_time,
            end_time=session.time_range.end_time,
            status=session.status.value,
            current_stage=session.current_stage.value if session.current_stage else None,
            progress=session.progress,
            trigger_type=session.trigger_type.value,
            trigger_source=session.trigger_source,
            metadata_=session.metadata,
            created_at=session.created_at,
            updated_at=session.updated_at,
            error_message=session.error_message
        )
    
    @staticmethod
    def to_domain(orm: DiagnosisSessionORM) -> DiagnosisSession:
        """ORM 转领域模型"""
        from domain.enums import DiagnosisStatus, DiagnosisStage, TriggerType
        
        service_identifier = ServiceIdentifier(
            service_name=orm.service_name,
            service_instance=orm.service_instance,
            environment=orm.environment
        )
        
        time_range = TimeRange(
            start_time=orm.start_time,
            end_time=orm.end_time
        )
        
        return DiagnosisSession(
            id=orm.id,
            session_id=orm.session_id,
            service_identifier=service_identifier,
            time_range=time_range,
            status=DiagnosisStatus(orm.status),
            current_stage=DiagnosisStage(orm.current_stage) if orm.current_stage else None,
            progress=orm.progress,
            trigger_type=TriggerType(orm.trigger_type),
            trigger_source=orm.trigger_source,
            metadata=orm.metadata_ or {},
            created_at=orm.created_at,
            updated_at=orm.updated_at,
            error_message=orm.error_message
        )

"""诊断会话仓储实现"""
from typing import Optional
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
import structlog

from domain.models.diagnosis_session import DiagnosisSession
from domain.repositories.diagnosis_session_repository import DiagnosisSessionRepository
from domain.enums import DiagnosisStatus
from ..models.diagnosis_orm import DiagnosisSessionORM
from ..mappers.diagnosis_mapper import DiagnosisMapper
from config.settings import settings

logger = structlog.get_logger(__name__)


class DiagnosisSessionRepositoryImpl(DiagnosisSessionRepository):
    """诊断会话仓储实现"""
    
    def __init__(self, session: Optional[AsyncSession] = None):
        self.session = session
        self.mapper = DiagnosisMapper()
    
    async def save(self, diagnosis_session: DiagnosisSession) -> None:
        """保存诊断会话"""
        if not settings.database_enabled or not self.session:
            logger.warning("database_disabled_skip_save", session_id=diagnosis_session.session_id)
            return
        
        try:
            orm = self.mapper.to_orm(diagnosis_session)
            self.session.add(orm)
            await self.session.flush()
            logger.info("diagnosis_session_saved", session_id=diagnosis_session.session_id)
        except Exception as e:
            logger.error("diagnosis_session_save_failed", session_id=diagnosis_session.session_id, error=str(e))
            raise
    
    async def find_by_id(self, session_id: str) -> Optional[DiagnosisSession]:
        """根据 ID 查找"""
        if not settings.database_enabled or not self.session:
            logger.warning("database_disabled_skip_find")
            return None
        
        try:
            result = await self.session.execute(
                select(DiagnosisSessionORM).where(DiagnosisSessionORM.session_id == session_id)
            )
            orm = result.scalar_one_or_none()
            
            if not orm:
                return None
            
            return self.mapper.to_domain(orm)
        except Exception as e:
            logger.error("diagnosis_session_find_failed", session_id=session_id, error=str(e))
            return None
    
    async def find_by_service(self, service_name: str, limit: int = 10) -> list[DiagnosisSession]:
        """根据服务名查找"""
        if not settings.database_enabled or not self.session:
            logger.warning("database_disabled_skip_find_by_service")
            return []
        
        try:
            result = await self.session.execute(
                select(DiagnosisSessionORM)
                .where(DiagnosisSessionORM.service_name == service_name)
                .order_by(DiagnosisSessionORM.created_at.desc())
                .limit(limit)
            )
            orms = result.scalars().all()
            return [self.mapper.to_domain(orm) for orm in orms]
        except Exception as e:
            logger.error("diagnosis_session_find_by_service_failed", service_name=service_name, error=str(e))
            return []
    
    async def find_by_status(self, status: DiagnosisStatus, limit: int = 10) -> list[DiagnosisSession]:
        """根据状态查找"""
        if not settings.database_enabled or not self.session:
            return []
        
        try:
            result = await self.session.execute(
                select(DiagnosisSessionORM)
                .where(DiagnosisSessionORM.status == status.value)
                .order_by(DiagnosisSessionORM.created_at.desc())
                .limit(limit)
            )
            orms = result.scalars().all()
            return [self.mapper.to_domain(orm) for orm in orms]
        except Exception as e:
            logger.error("diagnosis_session_find_by_status_failed", error=str(e))
            return []
    
    async def delete(self, session_id: str) -> bool:
        """删除诊断会话"""
        if not settings.database_enabled or not self.session:
            logger.warning("database_disabled_skip_delete")
            return False
        
        try:
            result = await self.session.execute(
                select(DiagnosisSessionORM).where(DiagnosisSessionORM.session_id == session_id)
            )
            orm = result.scalar_one_or_none()
            
            if not orm:
                return False
            
            await self.session.delete(orm)
            await self.session.flush()
            return True
        except Exception as e:
            logger.error("diagnosis_session_delete_failed", session_id=session_id, error=str(e))
            return False
    
    async def update(self, diagnosis_session: DiagnosisSession) -> None:
        """更新诊断会话"""
        await self.save(diagnosis_session)
```

"""Loki 日志查询客户端"""
import httpx
from typing import Optional, Any
from tenacity import retry, stop_after_attempt, wait_exponential
import structlog

from config.settings import settings

logger = structlog.get_logger(__name__)


class LokiClient:
    """Loki 日志查询客户端"""
    
    def __init__(self, base_url: Optional[str] = None, timeout: float = 30.0):
        self.base_url = base_url or settings.loki_url
        self.timeout = timeout
        self.enabled = settings.loki_enabled
        self._client: Optional[httpx.AsyncClient] = None
    
    async def __aenter__(self):
        if not self.enabled:
            logger.info("loki_disabled_skip_connect")
            return self
        
        self._client = httpx.AsyncClient(base_url=self.base_url, timeout=self.timeout)
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self._client:
            await self._client.aclose()
    
    @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=2, max=10), reraise=True)
    async def query_range(
        self,
        query: str,
        start: str,
        end: str,
        limit: int = 1000
    ) -> dict[str, Any]:
        """范围查询"""
        if not self.enabled or not self._client:
            logger.warning("loki_disabled_skip_query")
            return {"status": "disabled", "data": {"result": []}}
        
        params = {"query": query, "start": start, "end": end, "limit": limit}
        
        try:
            response = await self._client.get("/loki/api/v1/query_range", params=params)
            response.raise_for_status()
            return response.json()
        except Exception as e:
            logger.error("loki_query_failed", query=query, error=str(e))
            return {"status": "error", "error": str(e)}
    
    async def labels(self) -> dict[str, Any]:
        """获取标签列表"""
        if not self.enabled or not self._client:
            return {"status": "disabled", "data": []}
        
        try:
            response = await self._client.get("/loki/api/v1/labels")
            response.raise_for_status()
            return response.json()
        except Exception as e:
            logger.error("loki_labels_failed", error=str(e))
            return {"status": "error", "data": []}
    
    async def health(self) -> dict[str, bool]:
        """健康检查"""
        if not self.enabled:
            return {"is_healthy": False, "reason": "disabled"}
        
        try:
            if not self._client:
                self._client = httpx.AsyncClient(base_url=self.base_url, timeout=5.0)
            response = await self._client.get("/ready")
            return {"is_healthy": response.status_code == 200}
        except Exception:
            return {"is_healthy": False}

    async def search(
        self,
        service_name: Optional[str] = None,
        min_duration: Optional[str] = None,
        max_duration: Optional[str] = None,
        limit: int = 20
    ) -> dict[str, Any]:

            params["service.name"] = service_name
        if min_duration:
            params["minDuration"] = min_duration
        if max_duration:
            params["maxDuration"] = max_duration
        
        try:
            response = await self._client.get("/api/search", params=params)
            response.raise_for_status()
            return response.json()
        except Exception as e:
            logger.error("tempo_search_failed", error=str(e))
            return {"status": "error", "traces": []}
    
    async def health(self) -> dict[str, bool]:
        """健康检查"""
        if not self.enabled:
            return {"is_healthy": False, "reason": "disabled"}
        
        try:
            if not self._client:
                self._client = httpx.AsyncClient(base_url=self.base_url, timeout=5.0)
            response = await self._client.get("/status")
            return {"is_healthy": response.status_code == 200}
        except Exception:
            return {"is_healthy": False}
```

"""健康检查聚合器"""
from dataclasses import dataclass, field
from typing import Any
import structlog

from .prometheus_client import PrometheusClient
from .loki_client import LokiClient
from .tempo_client import TempoClient
from config.settings import settings

logger = structlog.get_logger(__name__)


@dataclass
class HealthStatus:
    """健康状态"""
    is_healthy: bool
    details: dict[str, dict[str, Any]] = field(default_factory=dict)
    
    def to_dict(self) -> dict[str, Any]:
        return {
            "is_healthy": self.is_healthy,
            "details": self.details
        }


class HealthChecker:
    """健康检查聚合器"""
    
    async def check_all(self) -> HealthStatus:
        """检查所有外部服务"""
        details = {}
        
        # 检查 Prometheus
        if settings.prometheus_enabled:
            try:
                async with PrometheusClient() as client:
                    prom_health = await client.health()
                    details["prometheus"] = {
                        "status": "healthy" if prom_health["is_healthy"] else "unhealthy",
                        "enabled": True
                    }
            except Exception as e:
                details["prometheus"] = {"status": "unhealthy", "enabled": True, "error": str(e)}
        else:
            details["prometheus"] = {"status": "disabled", "enabled": False}
        
        # 检查 Loki
        if settings.loki_enabled:
            try:
                async with LokiClient() as client:
                    loki_health = await client.health()
                    details["loki"] = {
                        "status": "healthy" if loki_health["is_healthy"] else "unhealthy",
                        "enabled": True
                    }
            except Exception as e:
                details["loki"] = {"status": "unhealthy", "enabled": True, "error": str(e)}
        else:
            details["loki"] = {"status": "disabled", "enabled": False}
        
        # 检查 Tempo
        if settings.tempo_enabled:
            try:
                async with TempoClient() as client:
                    tempo_health = await client.health()
                    details["tempo"] = {
                        "status": "healthy" if tempo_health["is_healthy"] else "unhealthy",
                        "enabled": True
                    }
            except Exception as e:
                details["tempo"] = {"status": "unhealthy", "enabled": True, "error": str(e)}
        else:
            details["tempo"] = {"status": "disabled", "enabled": False}
        
        # 检查数据库
        if settings.database_enabled:
            details["database"] = {"status": "enabled", "enabled": True}
        else:
            details["database"] = {"status": "disabled", "enabled": False}
        
        # 检查 Redis
        if settings.redis_enabled:
            details["redis"] = {"status": "enabled", "enabled": True}
        else:
            details["redis"] = {"status": "disabled", "enabled": False}
        
        # 检查 Kafka
        if settings.kafka_enabled:
            details["kafka"] = {"status": "enabled", "enabled": True}
        else:
            details["kafka"] = {"status": "disabled", "enabled": False}
        
        # 整体健康：至少核心服务可用
        is_healthy = True  # 默认健康（所有服务都可以禁用）
        
        return HealthStatus(is_healthy=is_healthy, details=details)
```
