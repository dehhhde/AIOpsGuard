```python
from pydantic import BaseModel
from typing import Generic, TypeVar, Optional

T = TypeVar('T')

class ApiResponse(BaseModel, Generic[T]):
    """统一 API 响应格式"""
    
    success: bool
    code: int
    message: str
    data: Optional[T] = None
    timestamp: datetime = Field(default_factory=datetime.now)

        """错误响应"""

            code=code,
            message=message,
            data=None,
```

#### 分页设计

```python
from pydantic import BaseModel
from typing import List, Generic, TypeVar

T = TypeVar('T')

class PageRequest(BaseModel):
    """分页请求"""
    page: int = Field(1, ge=1, description="页码")
    page_size: int = Field(10, ge=1, le=100, description="每页数量")
    sort_by: Optional[str] = Field(None, description="排序字段")
    order: str = Field("desc", regex="^(asc|desc)$", description="排序方向")


class PageResponse(BaseModel, Generic[T]):
    """分页响应"""
    items: List[T]
    total: int
    page: int
    page_size: int
    total_pages: int
    has_next: bool
    has_prev: bool
    
    @classmethod
    def from_query(
        cls,
        items: List[T],
        total: int,
        page: int,
        page_size: int
    ) -> "PageResponse[T]":
        """从查询结果创建分页响应"""
        total_pages = (total + page_size - 1) // page_size
        
        return cls(
            items=items,
            total=total,
            page=page,
            page_size=page_size,
            total_pages=total_pages,
            has_next=page < total_pages,
            has_prev=page > 1
        )


# 使用示例
@router.get("/diagnoses", response_model=ApiResponse[PageResponse[DiagnosisResponse]])
async def list_diagnoses(
    page_request: PageRequest = Depends(),
    service: Optional[str] = None
):
    """分页查询诊断列表"""
    items, total = await diagnosis_service.list_with_pagination(
        page=page_request.page,
        page_size=page_request.page_size,
        service=service,
        sort_by=page_request.sort_by,
        order=page_request.order
    )
    
    page_response = PageResponse.from_query(
        items=items,
        total=total,
        page=page_request.page,
        page_size=page_request.page_size
    )
    
    return ApiResponse.success_response(page_response)
```

```python
# 分表策略：按时间分表
class DiagnosisShardingStrategy:
    """诊断数据分表策略"""
    
    @staticmethod
    def get_table_name(created_at: datetime) -> str:
        """根据创建时间获取表名"""
        # 按月分表
        return f"diagnosis_reports_{created_at.strftime('%Y%m')}"
    
    @staticmethod
    def get_table_for_query(start_date: datetime, end_date: datetime) -> List[str]:
        """获取查询需要的所有表"""
        tables = []
        current = start_date.replace(day=1)
        
        while current <= end_date:
            tables.append(f"diagnosis_reports_{current.strftime('%Y%m')}")
            # 移到下个月
            if current.month == 12:
                current = current.replace(year=current.year + 1, month=1)
            else:
                current = current.replace(month=current.month + 1)
        
        return tables


# 使用示例
```

### 3.3 分布式锁

```python
import redis.asyncio as aioredis
import uuid
from contextlib import asynccontextmanager

class RedisDistributedLock:
    """Redis 分布式锁"""
    
    def __init__(self, redis_client: aioredis.Redis, key: str, timeout: int = 10):
        self.redis = redis_client
        self.key = f"lock:{key}"
        self.timeout = timeout
        self.identifier = str(uuid.uuid4())
    
    async def acquire(self) -> bool:
        """获取锁"""
        # SET key value NX EX timeout
        result = await self.redis.set(
            self.key,
            self.identifier,
            nx=True,  # 只在 key 不存在时设置
            ex=self.timeout
        )
        return result is not None
    
    async def release(self):
        """释放锁（Lua 脚本保证原子性）"""
        lua_script = """
        if redis.call("get", KEYS[1]) == ARGV[1] then
            return redis.call("del", KEYS[1])
        else
            return 0
        end
        """
        await self.redis.eval(lua_script, 1, self.key, self.identifier)
    
    @asynccontextmanager
    async def __call__(self):
        """上下文管理器"""
        acquired = await self.acquire()
        
        if not acquired:
            raise RuntimeError(f"Failed to acquire lock: {self.key}")
        
        try:
            yield
        finally:
            await self.release()


# 使用示例
async def process_with_lock(resource_id: str):
    """使用分布式锁处理资源"""
    lock = RedisDistributedLock(redis_client, f"resource:{resource_id}")
    
    async with lock():
        # 临界区代码
        resource = await get_resource(resource_id)
        await update_resource(resource)
```

## 4. 缓存设计

### 4.1 多级缓存

```python
from typing import Optional, Any
import pickle

class MultiLevelCache:
    """
    多级缓存
    
    L1: 内存缓存（进程内）
    L2: Redis 缓存（分布式）
    L3: 数据库（持久化）
    """
    
    def __init__(
        self,
        redis_client: aioredis.Redis,
        l1_max_size: int = 1000
    ):
        self.redis = redis_client
        self.l1_cache: dict = {}  # 简化的 L1 缓存
        self.l1_max_size = l1_max_size
    
    async def get(self, key: str) -> Optional[Any]:
        """获取缓存"""
        # 1. 尝试 L1 缓存
        if key in self.l1_cache:
            return self.l1_cache[key]
        
        # 2. 尝试 L2 缓存（Redis）
        value = await self.redis.get(key)
        if value:
            # 反序列化
            deserialized = pickle.loads(value)
            
            # 写入 L1 缓存
            self._set_l1(key, deserialized)
            
            return deserialized
        
        # 3. 缓存未命中，返回 None
        return None
    
    async def set(
        self,
        key: str,
        value: Any,
        ttl: int = 3600
    ):
        """设置缓存"""
        # 1. 写入 L1 缓存
        self._set_l1(key, value)
        
        # 2. 写入 L2 缓存（Redis）
        serialized = pickle.dumps(value)
        await self.redis.set(key, serialized, ex=ttl)
    
    def _set_l1(self, key: str, value: Any):
        """设置 L1 缓存（LRU 淘汰）"""
        if len(self.l1_cache) >= self.l1_max_size:
            # 简单的 FIFO 淘汰（实际应用中使用 functools.lru_cache）
            first_key = next(iter(self.l1_cache))
            del self.l1_cache[first_key]
        
        self.l1_cache[key] = value
    
    async def delete(self, key: str):
        """删除缓存"""
        # 删除 L1
        self.l1_cache.pop(key, None)
        
        # 删除 L2
        await self.redis.delete(key)
```

```python
class SingleFlightCache:
    """单次飞行缓存（防止击穿）"""
    
    def __init__(self):
        self.loading = {}  # key -> Future
        self.lock = asyncio.Lock()
    
    async def get(self, key: str, loader: Callable):
        """获取缓存（防击穿）"""
        # 1. 尝试缓存
        value = await redis.get(key)
        if value:
            return value
        
        # 2. 检查是否有其他协程正在加载
        async with self.lock:
            if key in self.loading:
                # 等待其他协程加载完成
                return await self.loading[key]
            
            # 创建加载任务
            future = asyncio.create_task(self._load_and_cache(key, loader))
            self.loading[key] = future
        
        try:
            return await future
        finally:
            async with self.lock:
                self.loading.pop(key, None)
    
    async def _load_and_cache(self, key: str, loader: Callable):
        """加载并缓存"""
        value = await loader()
        await redis.set(key, value, ex=3600)
        return value
```

```python
class DelayedQueue:
    """延迟队列（基于 Redis Sorted Set）"""
    
    def __init__(self, redis_client: aioredis.Redis, queue_name: str):
        self.redis = redis_client
        self.queue_name = f"delayed_queue:{queue_name}"
        self.is_running = False
    
    async def push(self, message: dict, delay_seconds: int):
        """推送延迟消息"""
        # 计算触发时间
        trigger_time = time.time() + delay_seconds
        
        # 添加到 Sorted Set
        await self.redis.zadd(
            self.queue_name,
            {json.dumps(message): trigger_time}
        )
    
    async def start_worker(self, handler: Callable):
        """启动延迟队列工作线程"""
        self.is_running = True
        
        while self.is_running:
            try:
                # 1. 获取到期的消息
                now = time.time()
                messages = await self.redis.zrangebyscore(
                    self.queue_name,
                    min=0,
                    max=now,
                    start=0,
                    num=10
                )
                
                for message_json in messages:
                    message = json.loads(message_json)
                    
                    # 2. 处理消息
                    try:
                        await handler(message)
                    except Exception as e:
                        logging.error(f"Failed to process delayed message: {e}")
                    
                    # 3. 从队列中移除
                    await self.redis.zrem(self.queue_name, message_json)
                
                # 休眠 1 秒
                await asyncio.sleep(1)
            
            except Exception as e:
                logging.error(f"Delayed queue worker error: {e}")
                await asyncio.sleep(5)
    
    def stop(self):
        """停止工作线程"""
        self.is_running = False
```

```python
import time
from collections import defaultdict

class RateLimiter:
    """令牌桶限流器"""
    
    def __init__(
        self,
        rate: int = 100,  # 每秒令牌数
        capacity: int = 100  # 桶容量
    ):
        self.rate = rate
        self.capacity = capacity
        self.buckets = defaultdict(lambda: {
            "tokens": capacity,
            "last_update": time.time()
        })
    
    def allow(self, key: str) -> bool:
        """检查是否允许请求"""
        now = time.time()
        bucket = self.buckets[key]
        
        # 1. 计算新增令牌
        elapsed = now - bucket["last_update"]
        new_tokens = elapsed * self.rate
        
        # 2. 更新令牌数（不超过容量）
        bucket["tokens"] = min(
            self.capacity,
            bucket["tokens"] + new_tokens
        )
        bucket["last_update"] = now
        
        # 3. 检查是否有令牌
        if bucket["tokens"] >= 1:
            bucket["tokens"] -= 1
            return True
        
        return False


# Redis 实现的分布式限流
class RedisRateLimiter:
    """基于 Redis 的分布式限流"""
    
    def __init__(self, redis_client: aioredis.Redis):
        self.redis = redis_client
    
    async def allow(
        self,
        key: str,
        limit: int = 100,
        window: int = 60
    ) -> bool:
        """
        滑动窗口限流
        
        :param key: 限流键
        :param limit: 限制次数
        :param window: 时间窗口（秒）
        """
        now = time.time()
        window_start = now - window
        
        # Lua 脚本保证原子性
        lua_script = """
        local key = KEYS[1]
        local now = tonumber(ARGV[1])
        local window_start = tonumber(ARGV[2])
        local limit = tonumber(ARGV[3])
        
        -- 删除过期数据
        redis.call('ZREMRANGEBYSCORE', key, 0, window_start)
        
        -- 获取当前计数
        local count = redis.call('ZCARD', key)
        
        if count < limit then
            -- 添加当前请求
            redis.call('ZADD', key, now, now)
            redis.call('EXPIRE', key, 60)
            return 1
        else
            return 0
        end
        """
        
        result = await self.redis.eval(
            lua_script,
            1,  # keys 数量
            f"rate_limit:{key}",  # KEYS[1]
            now,  # ARGV[1]
            window_start,  # ARGV[2]
            limit  # ARGV[3]
        )
        
        return result == 1
```

```python
from enum import Enum
import time

class CircuitState(Enum):
    """熔断器状态"""
    CLOSED = "closed"      # 正常
    OPEN = "open"          # 熔断
    HALF_OPEN = "half_open"  # 半开（探测）


class CircuitBreaker:
    """熔断器"""
    
    def __init__(
        self,
        failure_threshold: int = 5,      # 失败阈值
        success_threshold: int = 2,      # 成功阈值（半开->关闭）
        timeout: int = 60,               # 熔断超时（秒）
    ):
        self.failure_threshold = failure_threshold
        self.success_threshold = success_threshold
        self.timeout = timeout
        
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.success_count = 0
        self.last_failure_time = None
    
    def call(self, func: Callable, *args, **kwargs):
        """执行调用（带熔断保护）"""
        # 检查熔断器状态
        self._check_state()
        
        if self.state == CircuitState.OPEN:
            raise Exception("Circuit breaker is OPEN")
        
        try:
            # 执行调用
            result = func(*args, **kwargs)
            
            # 记录成功
            self._on_success()
            
            return result
        
        except Exception as e:
            # 记录失败
            self._on_failure()
            raise
    
    def _check_state(self):
        """检查并更新状态"""
        if self.state == CircuitState.OPEN:
            # 检查是否超过超时时间
            if time.time() - self.last_failure_time > self.timeout:
                # 进入半开状态
                self.state = CircuitState.HALF_OPEN
                self.success_count = 0
                logging.info("Circuit breaker: OPEN -> HALF_OPEN")
    
    def _on_success(self):
        """成功回调"""
        self.failure_count = 0
        
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            
            # 达到成功阈值，关闭熔断器
            if self.success_count >= self.success_threshold:
                self.state = CircuitState.CLOSED
                logging.info("Circuit breaker: HALF_OPEN -> CLOSED")
    
    def _on_failure(self):
        """失败回调"""
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.state == CircuitState.HALF_OPEN:
            # 半开状态失败，立即打开
            self.state = CircuitState.OPEN
            logging.warning("Circuit breaker: HALF_OPEN -> OPEN")
        
        elif self.failure_count >= self.failure_threshold:
            # 达到失败阈值，打开熔断器
            self.state = CircuitState.OPEN
            logging.warning("Circuit breaker: CLOSED -> OPEN")


# 使用示例

    failure_threshold=5,
    success_threshold=2,
    timeout=60
)
```

```python
import asyncio
from typing import Callable, Type
from functools import wraps

class RetryStrategy:
    """重试策略"""
    
    @staticmethod
    def fixed_delay(delay: float):
        """固定延迟"""
        return lambda attempt: delay
    
    @staticmethod
    def exponential_backoff(base: float = 1.0, multiplier: float = 2.0):
        """指数退避"""
        return lambda attempt: base * (multiplier ** attempt)
    
    @staticmethod
    def linear_backoff(base: float = 1.0, increment: float = 1.0):
        """线性退避"""
        return lambda attempt: base + (increment * attempt)


def retry(
    max_attempts: int = 3,
    delay_strategy: Callable[[int], float] = RetryStrategy.exponential_backoff(),
    exceptions: tuple = (Exception,)
):
    """重试装饰器"""
    def decorator(func: Callable):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            last_exception = None
            
            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                
                except exceptions as e:
                    last_exception = e
                    
                    if attempt < max_attempts - 1:
                        # 计算延迟
                        delay = delay_strategy(attempt)
                        
                        logging.warning(
                            f"Attempt {attempt + 1}/{max_attempts} failed: {e}. "
                            f"Retrying in {delay}s..."
                        )
                        
                        await asyncio.sleep(delay)
                    else:
                        logging.error(
                            f"All {max_attempts} attempts failed. Giving up."
                        )
            
            raise last_exception
        
        return wrapper
    return decorator


# 使用示例
```

```python
class FallbackStrategy:
    """降级策略"""
    
    @staticmethod
    async def with_fallback(
        primary: Callable,
        fallback: Callable,
        exceptions: tuple = (Exception,)
    ):
        """主备切换"""
        try:
            return await primary()
        except exceptions as e:
            logging.warning(f"Primary failed: {e}. Using fallback.")
            return await fallback()
    
    @staticmethod
    async def with_cache_fallback(
        loader: Callable,
        cache_key: str,
        redis: aioredis.Redis
    ):
        """缓存降级"""
        try:
            # 尝试加载新数据
            data = await loader()
            
            # 更新缓存
            await redis.set(cache_key, json.dumps(data), ex=3600)
            
            return data
        
        except Exception as e:
            # 失败时使用缓存
            logging.warning(f"Loader failed: {e}. Using cached data.")
            
            cached = await redis.get(cache_key)
            if cached:
                return json.loads(cached)
            
            # 缓存也没有，返回默认值
            return None
    
    @staticmethod
    async def with_default_value(
        func: Callable,
        default_value: Any
    ):
        """默认值降级"""
        try:
            return await func()
        except Exception as e:
            logging.warning(f"Function failed: {e}. Using default value.")
            return default_value
```

```python
class ShardedCache:
    """分片缓存（提升并发性能）"""
    
    def __init__(self, redis_clients: List[aioredis.Redis]):
        self.clients = redis_clients
        self.shard_count = len(redis_clients)

        return await client.get(key)
```

```python
    async def limited_task(task):
        async with semaphore:
            return await task
    
    results = await asyncio.gather(*[limited_task(t) for t in tasks])
    
    return results
```
