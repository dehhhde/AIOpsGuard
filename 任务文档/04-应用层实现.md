```python
"""请求 DTO"""
from pydantic import BaseModel, Field
from datetime import datetime
from typing import Optional


class TriggerDiagnosisRequest(BaseModel):
    """触发诊断请求
    
    Attributes:
        service_name: 服务名称
        service_instance: 服务实例（可选）
        environment: 环境（prod/staging/dev）
        time_range_minutes: 时间范围（分钟），默认最近 60 分钟
        trigger_source: 触发来源（manual/alert/scheduled）
    """
    service_name: str = Field(..., min_length=1, max_length=200, description="服务名称")
    service_instance: Optional[str] = Field(None, max_length=200, description="服务实例")
    environment: str = Field(default="prod", description="环境")
    time_range_minutes: int = Field(default=60, ge=1, le=1440, description="时间范围（分钟）")
    trigger_source: str = Field(default="manual", description="触发来源")
    
    class Config:
        json_schema_extra = {
            "example": {
                "service_name": "user-service",
                "service_instance": "user-service-01",
                "environment": "prod",
                "time_range_minutes": 60,
                "trigger_source": "manual"
            }
        }


class QueryDiagnosisRequest(BaseModel):
    """查询诊断请求"""
    session_id: Optional[str] = Field(None, description="诊断会话 ID")
    service_name: Optional[str] = Field(None, description="服务名称")
    status: Optional[str] = Field(None, description="诊断状态")
    limit: int = Field(default=10, ge=1, le=100, description="返回数量限制")
```

```python
from pydantic import BaseModel, Field
from datetime import datetime
from typing import Optional, List

    service_name: str

    environment: str
    status: str
    current_stage: Optional[str]

    end_time: datetime

        from_attributes = True

    """证据响应"""
    id: str
    evidence_type: str

    confidence: int
    timestamp: datetime

    """根因响应"""
    id: str
    problem_type: str
    summary: str
    reasoning: str
    confidence: int

    root_causes: List[RootCauseResponse]

    class Config:
        from_attributes = True

    """触发诊断响应"""
```

#### 4.2.1 触发诊断用例

**application/use_cases/trigger_diagnosis_use_case.py**

```python
"""触发诊断用例"""
import asyncio
from datetime import datetime, timedelta
import structlog

from domain.models.diagnosis_session import DiagnosisSession
from domain.value_objects.time_range import TimeRange
from domain.value_objects.service_identifier import ServiceIdentifier
from domain.enums import TriggerType
from infrastructure.cache import DistributedLock, RedisClient
from application.dto.requests import TriggerDiagnosisRequest
from application.dto.responses import TriggerDiagnosisResponse

logger = structlog.get_logger(__name__)


class DiagnosisAlreadyInProgressError(Exception):
    """诊断已在进行中异常"""
    pass


class TriggerDiagnosisUseCase:
    """触发诊断用例
    
    职责：
    1. 验证请求参数
    2. 使用分布式锁防止重复诊断
    3. 创建诊断会话
    4. 异步启动诊断流程
    """
    
    def __init__(self, redis_client: RedisClient):
        self.lock = DistributedLock(redis_client)
    
    async def execute(
        self,
        request: TriggerDiagnosisRequest
    ) -> TriggerDiagnosisResponse:
        """执行用例
        
        Args:
            request: 触发诊断请求
        
        Returns:
            触发诊断响应
        
        Raises:
            DiagnosisAlreadyInProgressError: 服务正在诊断中
        """
        service_name = request.service_name
        lock_key = f"diagnosis:{service_name}"
        
        logger.info(
            "trigger_diagnosis_start",
            service_name=service_name,
            environment=request.environment
        )
        
        # 尝试获取分布式锁
        try:
            async with self.lock.lock(
                resource=lock_key,
                ttl=300,  # 5 分钟
                wait=False
            ):
                # 创建诊断会话
                session = await self._create_session(request)
                
                # 异步启动诊断（不阻塞响应）
                asyncio.create_task(self._run_diagnosis_async(session))
                
                logger.info(
                    "diagnosis_triggered",
                    session_id=session.session_id,
                    service_name=service_name
                )
                
                return TriggerDiagnosisResponse(
                    session_id=session.session_id,
                    status="triggered",
                    message=f"Diagnosis for {service_name} has been triggered"
                )
                
        except RuntimeError:
            # 锁获取失败，服务正在诊断中
            logger.warning(
                "diagnosis_already_in_progress",
                service_name=service_name
            )
            raise DiagnosisAlreadyInProgressError(
                f"Service {service_name} is already being diagnosed. "
                f"Please wait for the current diagnosis to complete."
            )
    
    async def _create_session(
        self,
        request: TriggerDiagnosisRequest
    ) -> DiagnosisSession:
        """创建诊断会话"""
        # 构建时间范围
        end_time = datetime.now()
        start_time = end_time - timedelta(minutes=request.time_range_minutes)
        time_range = TimeRange(start_time=start_time, end_time=end_time)
        
        # 构建服务标识
        service_identifier = ServiceIdentifier(
            service_name=request.service_name,
            service_instance=request.service_instance,
            environment=request.environment
        )
        
        # 创建诊断会话
        session = DiagnosisSession(
            session_id=f"diag-{datetime.now().strftime('%Y%m%d%H%M%S')}-{request.service_name}",
            service_identifier=service_identifier,
            time_range=time_range,
            trigger_type=TriggerType(request.trigger_source),
            trigger_source=f"api:{request.trigger_source}"
        )
        
        return session
    
    async def _run_diagnosis_async(self, session: DiagnosisSession):
        """异步执行诊断（后台任务）"""
        try:
            logger.info(
                "diagnosis_execution_start",
                session_id=session.session_id
            )
            
            # TODO: 调用诊断服务（阶段五实现）
            # await diagnosis_service.run(session)
            
            # 目前只是占位
            await asyncio.sleep(1)
            
            logger.info(
                "diagnosis_execution_placeholder",
                session_id=session.session_id,
                message="Diagnosis service will be implemented in Stage 5"
            )
            
        except Exception as e:
            logger.error(
                "diagnosis_execution_failed",
                session_id=session.session_id,
                error=str(e)
            )
```

#### 4.2.2 查询诊断用例

**application/use_cases/query_diagnosis_use_case.py**

```python
"""查询诊断用例"""
from typing import Optional, List
import structlog

from domain.models.diagnosis_session import DiagnosisSession
from domain.repositories.diagnosis_session_repository import DiagnosisSessionRepository
from domain.enums import DiagnosisStatus
from infrastructure.persistence.database import get_db_session
from infrastructure.persistence.repositories import DiagnosisSessionRepositoryImpl
from infrastructure.cache import SessionCache, RedisClient
from application.dto.requests import QueryDiagnosisRequest
from application.dto.responses import DiagnosisSessionResponse
from application.mappers.diagnosis_dto_mapper import DiagnosisDTOMapper

logger = structlog.get_logger(__name__)


class QueryDiagnosisUseCase:
    """查询诊断用例
    
    职责：
    1. 先查缓存
    2. 缓存未命中查数据库
    3. 转换为响应 DTO
    """
    
    def __init__(self, redis_client: RedisClient):
        self.cache = SessionCache(redis_client)
        self.mapper = DiagnosisDTOMapper()
    
    async def execute(
        self,
        request: QueryDiagnosisRequest
    ) -> List[DiagnosisSessionResponse]:
        """执行查询
        
        Args:
            request: 查询请求
        
        Returns:
            诊断会话响应列表
        """
        # 如果指定了 session_id，查询单个
        if request.session_id:
            session = await self._find_by_id(request.session_id)
            return [self.mapper.to_session_response(session)] if session else []
        
        # 如果指定了 service_name，查询服务的诊断历史
        if request.service_name:
            sessions = await self._find_by_service(
                request.service_name,
                request.limit
            )
            return [
                self.mapper.to_session_response(s)
                for s in sessions
            ]
        
        # 如果指定了 status，查询指定状态的诊断
        if request.status:
            status = DiagnosisStatus(request.status)
            sessions = await self._find_by_status(status, request.limit)
            return [
                self.mapper.to_session_response(s)
                for s in sessions
            ]
        
        # 默认返回空列表
        return []
    
    async def _find_by_id(self, session_id: str) -> Optional[DiagnosisSession]:
        """根据 ID 查找（先缓存后数据库）"""
        # 1. 先查缓存
        cached_data = await self.cache.get(session_id)
        if cached_data:
            logger.debug("session_found_in_cache", session_id=session_id)
            # TODO: 从缓存数据重建领域模型
            # 目前直接查数据库
        
        # 2. 查数据库
        async with get_db_session() as db_session:
            if not db_session:
                logger.warning("database_disabled_skip_query")
                return None
            
            repo = DiagnosisSessionRepositoryImpl(db_session)
            return await repo.find_by_id(session_id)
    
    async def _find_by_service(
        self,
        service_name: str,
        limit: int
    ) -> List[DiagnosisSession]:
        """根据服务名查找"""
        async with get_db_session() as db_session:
            if not db_session:
                return []
            
            repo = DiagnosisSessionRepositoryImpl(db_session)
            return await repo.find_by_service(service_name, limit)
    
    async def _find_by_status(
        self,
        status: DiagnosisStatus,
        limit: int
    ) -> List[DiagnosisSession]:
        """根据状态查找"""
        async with get_db_session() as db_session:
            if not db_session:
                return []
            
            repo = DiagnosisSessionRepositoryImpl(db_session)
            return await repo.find_by_status(status, limit)
```

```python
"""诊断应用服务"""
import structlog
from typing import List

from application.use_cases.trigger_diagnosis_use_case import (
    TriggerDiagnosisUseCase,
    DiagnosisAlreadyInProgressError
)
from application.use_cases.query_diagnosis_use_case import QueryDiagnosisUseCase
from application.dto.requests import TriggerDiagnosisRequest, QueryDiagnosisRequest
from application.dto.responses import (
    TriggerDiagnosisResponse,
    DiagnosisSessionResponse
)
from infrastructure.cache import RedisClient

logger = structlog.get_logger(__name__)


class DiagnosisApplicationService:
    """诊断应用服务
    
    职责：
    1. 协调多个用例
    2. 统一异常处理
    3. 事务边界管理
    """
    
    def __init__(self, redis_client: RedisClient):
        self.trigger_use_case = TriggerDiagnosisUseCase(redis_client)
        self.query_use_case = QueryDiagnosisUseCase(redis_client)
    
    async def trigger_diagnosis(
        self,
        request: TriggerDiagnosisRequest
    ) -> TriggerDiagnosisResponse:
        """触发诊断
        
        Args:
            request: 触发诊断请求
        
        Returns:
            触发诊断响应
        
        Raises:
            DiagnosisAlreadyInProgressError: 服务正在诊断中
        """
        return await self.trigger_use_case.execute(request)
    
    async def query_diagnosis(
        self,
        request: QueryDiagnosisRequest
    ) -> List[DiagnosisSessionResponse]:
        """查询诊断
        
        Args:
            request: 查询请求
        
        Returns:
            诊断会话响应列表
        """
        return await self.query_use_case.execute(request)
    
    async def get_diagnosis_by_id(
        self,
        session_id: str
    ) -> DiagnosisSessionResponse | None:
        """根据 ID 获取诊断
        
        Args:
            session_id: 诊断会话 ID
        
        Returns:
            诊断会话响应，如果不存在返回 None
        """
        request = QueryDiagnosisRequest(session_id=session_id)
        results = await self.query_use_case.execute(request)
        return results[0] if results else None
```

```python
"""依赖注入容器"""
from infrastructure.cache import RedisClient
from infrastructure.persistence.database import init_engine
from application.services.diagnosis_application_service import DiagnosisApplicationService


class Container:
    """依赖注入容器
    
    职责：管理应用级别的依赖
    """
    
    def __init__(self):
        self._redis_client: RedisClient | None = None
        self._diagnosis_service: DiagnosisApplicationService | None = None
    
    async def init(self):
        """初始化容器"""
        # 初始化数据库引擎
        init_engine()
        
        # 初始化 Redis 客户端
        self._redis_client = RedisClient()
        await self._redis_client.connect()
        
        # 初始化应用服务
        self._diagnosis_service = DiagnosisApplicationService(self._redis_client)
    
    async def close(self):
        """关闭容器"""
        if self._redis_client:
            await self._redis_client.close()
    
    @property
    def diagnosis_service(self) -> DiagnosisApplicationService:
        """获取诊断应用服务"""
        if not self._diagnosis_service:
            raise RuntimeError("Container not initialized")
        return self._diagnosis_service
    
    @property
    def redis_client(self) -> RedisClient:
        """获取 Redis 客户端"""
        if not self._redis_client:
            raise RuntimeError("Container not initialized")
        return self._redis_client


# 全局容器实例
container = Container()
```
