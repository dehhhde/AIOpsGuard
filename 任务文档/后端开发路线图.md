```python
class PostgresSessionRepository(DiagnosisSessionRepository):
    async def save(self, session: DiagnosisSession) -> None:
        # 1. 将领域模型映射为 ORM 模型
        model = DiagnosisMapper.to_orm(session)
        # 2. 保存到数据库
        self.db_session.add(model)
        await self.db_session.commit()
    
    async def find_by_id(self, diagnosis_id: str) -> DiagnosisSession | None:
        # 1. 从数据库查询
        model = await self.db_session.get(DiagnosisSessionModel, diagnosis_id)
        # 2. 映射为领域模型
        return DiagnosisMapper.to_domain(model) if model else None
```

```python
class AnomalyDetector:
    def threshold_detect(self, values: list[float], threshold: float) -> bool:
        """固定阈值检测"""
        return any(v > threshold for v in values)
    
    def sigma_detect(self, values: list[float], sigma: float = 3.0) -> bool:
        """3-sigma 检测"""
        mean = statistics.mean(values)
        stdev = statistics.stdev(values)
        return any(abs(v - mean) > sigma * stdev for v in values)
    
    def compare_detect(self, current: float, baseline: float, threshold: float = 0.5) -> bool:
        """环比检测（增幅 > 50%）"""
        return (current - baseline) / baseline > threshold
```

```python
class RCAAgent:
    async def analyze(
        self,
        session: DiagnosisSession,
        evidence_list: list[Evidence]
    ) -> list[RootCause]:
        # 1. 构建 Prompt
        prompt = self.prompt_builder.build(session, evidence_list)
        
        # 2. 调用 LLM
        response = await self.llm_client.complete(prompt)
        
        # 3. 解析响应
        causes = self.parser.parse(response)
        
        # 4. 计算置信度
        root_causes = []
        for rank, cause in enumerate(causes, 1):
            confidence = self.calculate_confidence(cause, evidence_list)
            root_causes.append(RootCause(
                rank=rank,
                cause=cause.description,
                confidence=confidence,
                evidence_ids=[e.id for e in cause.related_evidence]
            ))
        
        return root_causes
```
