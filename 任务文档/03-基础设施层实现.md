```python
"""Prometheus 客户端"""
import httpx
from typing import Optional, Any
from tenacity import retry, stop_after_attempt, wait_exponential

from config.settings import settings


class PrometheusClient:
    """Prometheus 查询客户端
    
    职责：
    1. 即时查询（query）
    2. 范围查询（query_range）
    3. 序列查询（series）
    4. 健康检查
    
    特性：
    - 异步 HTTP 客户端
    - 自动重试
    - 超时控制
    - 连接池管理
    """
    
    def __init__(
        self,
        base_url: Optional[str] = None,
        timeout: float = 30.0
    ):
        """初始化客户端
        
        Args:
            base_url: Prometheus 地址，默认从配置读取
            timeout: 请求超时（秒）
        """
        self.base_url = base_url or settings.prometheus_url
        self.timeout = timeout
        self._client: Optional[httpx.AsyncClient] = None
    
    async def __aenter__(self):
        """上下文管理器入口"""
        self._client = httpx.AsyncClient(
            base_url=self.base_url,
            timeout=self.timeout,
            limits=httpx.Limits(max_connections=10, max_keepalive_connections=5)
        )
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """上下文管理器退出"""
        if self._client:
            await self._client.aclose()
    
    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=2, max=10)
    )
    async def query(
        self,
        query: str,
        time: Optional[str] = None
    ) -> dict[str, Any]:
        """即时查询
        
        Args:
            query: PromQL 查询表达式
            time: 查询时间（RFC3339 或 Unix 时间戳），默认当前时间
        
        Returns:
            查询结果
        
        Examples:
            >>> async with PrometheusClient() as client:
            ...     result = await client.query("up")
            ...     print(result["status"])  # "success"
        """
        params = {"query": query}
        if time:
            params["time"] = time
        
        response = await self._client.get("/api/v1/query", params=params)
        response.raise_for_status()
        return response.json()
    
    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=2, max=10)
    )
    async def query_range(
        self,
        query: str,
        start: str,
        end: str,
        step: str = "15s"
    ) -> dict[str, Any]:
        """范围查询
        
        Args:
            query: PromQL 查询表达式
            start: 开始时间（RFC3339 或 Unix 时间戳）
            end: 结束时间（RFC3339 或 Unix 时间戳）
            step: 查询步长（如 15s, 1m, 5m）
        
        Returns:
            查询结果
        
        Examples:
            >>> result = await client.query_range(
            ...     query="rate(requests_total[5m])",
            ...     start="2025-01-09T10:00:00Z",
            ...     end="2025-01-09T11:00:00Z",
            ...     step="1m"
            ... )
        """
        params = {
            "query": query,
            "start": start,
            "end": end,
            "step": step
        }
        
        response = await self._client.get("/api/v1/query_range", params=params)
        response.raise_for_status()
        return response.json()
    
    async def series(
        self,
        match: list[str],
        start: Optional[str] = None,
        end: Optional[str] = None
    ) -> dict[str, Any]:
        """序列查询
        
        查询匹配的时间序列标签
        
        Args:
            match: 匹配表达式列表
            start: 开始时间
            end: 结束时间
        
        Returns:
            序列列表
        """
        params = {"match[]": match}
        if start:
            params["start"] = start
        if end:
            params["end"] = end
        
        response = await self._client.get("/api/v1/series", params=params)
        response.raise_for_status()
        return response.json()
    
    async def health(self) -> dict[str, bool]:
        """健康检查
        
        Returns:
            {"is_healthy": True/False}
        """
        try:
            response = await self._client.get("/-/healthy")
            return {"is_healthy": response.status_code == 200}
        except Exception:
            return {"is_healthy": False}
```

```python
"""健康检查聚合器"""
from typing import Any
from dataclasses import dataclass, field

from .prometheus_client import PrometheusClient
from .loki_client import LokiClient
from .tempo_client import TempoClient


@dataclass
class HealthStatus:
    """健康状态"""
    is_healthy: bool
    details: dict[str, dict[str, Any]] = field(default_factory=dict)
    
    def to_dict(self) -> dict[str, Any]:
        return {
            "is_healthy": self.is_healthy,
            "details": self.details
        }


class HealthChecker:
    """健康检查聚合器
    
    职责：检查所有外部依赖的健康状态
    """
    
    async def check_all(self) -> HealthStatus:
        """检查所有外部服务
        
        Returns:
            聚合的健康状态
        """
        details = {}
        
        # 检查 Prometheus
        try:
            async with PrometheusClient() as client:
                prom_health = await client.health()
                details["prometheus"] = {
                    "status": "healthy" if prom_health["is_healthy"] else "unhealthy",
                    "message": "OK" if prom_health["is_healthy"] else "Connection failed"
                }
        except Exception as e:
            details["prometheus"] = {
                "status": "unhealthy",
                "message": str(e)
            }
        
        # 检查 Loki（可选）
        try:
            async with LokiClient() as client:
                loki_health = await client.health()
                details["loki"] = {
                    "status": "healthy" if loki_health["is_healthy"] else "unhealthy",
                    "message": "OK" if loki_health["is_healthy"] else "Connection failed"
                }
        except Exception:
            details["loki"] = {
                "status": "unavailable",
                "message": "Not configured or unreachable"
            }
        
        # 检查 Tempo（可选）
        try:
            async with TempoClient() as client:
                tempo_health = await client.health()
                details["tempo"] = {
                    "status": "healthy" if tempo_health["is_healthy"] else "unhealthy",
                    "message": "OK" if tempo_health["is_healthy"] else "Connection failed"
                }
        except Exception:
            details["tempo"] = {
                "status": "unavailable",
                "message": "Not configured or unreachable"
            }
        
        # 至少 Prometheus 健康才算整体健康
        is_healthy = details.get("prometheus", {}).get("status") == "healthy"
        
        return HealthStatus(is_healthy=is_healthy, details=details)
```

```python
"""数据库连接管理"""
from sqlalchemy.ext.asyncio import (
    create_async_engine,
    AsyncSession,
    async_sessionmaker
)
from sqlalchemy.orm import declarative_base
from contextlib import asynccontextmanager

from config.settings import settings

# 创建异步引擎
engine = create_async_engine(
    settings.database_url,
    echo=settings.database_echo,
    pool_size=settings.database_pool_size,
    max_overflow=settings.database_max_overflow,
    pool_pre_ping=True  # 连接前检查健康
)

# 创建会话工厂
AsyncSessionLocal = async_sessionmaker(
    engine,
    class_=AsyncSession,
    expire_on_commit=False
)

# 声明基类
Base = declarative_base()


@asynccontextmanager
async def get_db_session():
    """获取数据库会话
    
    使用上下文管理器自动管理事务
    
    Examples:
        >>> async with get_db_session() as session:
        ...     result = await session.execute(select(User))
        ...     users = result.scalars().all()
    """
    async with AsyncSessionLocal() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()


async def init_db():
    """初始化数据库（创建表）"""
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)


async def close_db():
    """关闭数据库连接"""
    await engine.dispose()
```

```python
"""诊断会话仓储实现"""
from typing import Optional
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from domain.models.diagnosis_session import DiagnosisSession
from domain.repositories.diagnosis_session_repository import DiagnosisSessionRepository
from domain.enums import DiagnosisStatus
from ..models.diagnosis_orm import DiagnosisSessionORM
from ..mappers.diagnosis_mapper import DiagnosisMapper


class DiagnosisSessionRepositoryImpl(DiagnosisSessionRepository):
    """诊断会话仓储实现"""
    
    def __init__(self, session: AsyncSession):
        self.session = session
        self.mapper = DiagnosisMapper()
    
    async def save(self, diagnosis_session: DiagnosisSession) -> None:
        """保存诊断会话"""
        # 转换为 ORM 模型
        orm = self.mapper.to_orm(diagnosis_session)
        
        # 合并或添加
        self.session.add(orm)
        await self.session.flush()
    
    async def find_by_id(self, session_id: str) -> Optional[DiagnosisSession]:
        """根据 ID 查找"""
        result = await self.session.execute(
            select(DiagnosisSessionORM).where(
                DiagnosisSessionORM.session_id == session_id
            )
        )
        orm = result.scalar_one_or_none()
        
        if not orm:
            return None
        
        return self.mapper.to_domain(orm)
    
    async def find_by_service(
        self,
        service_name: str,
        limit: int = 10
    ) -> list[DiagnosisSession]:
        """根据服务名查找"""
        result = await self.session.execute(
            select(DiagnosisSessionORM)
            .where(DiagnosisSessionORM.service_name == service_name)
            .order_by(DiagnosisSessionORM.created_at.desc())
            .limit(limit)
        )
        orms = result.scalars().all()
        
        return [self.mapper.to_domain(orm) for orm in orms]
    
    async def find_by_status(
        self,
        status: DiagnosisStatus,
        limit: int = 10
    ) -> list[DiagnosisSession]:
        """根据状态查找"""
        result = await self.session.execute(
            select(DiagnosisSessionORM)
            .where(DiagnosisSessionORM.status == status)
            .order_by(DiagnosisSessionORM.created_at.desc())
            .limit(limit)
        )
        orms = result.scalars().all()
        
        return [self.mapper.to_domain(orm) for orm in orms]
    
    async def delete(self, session_id: str) -> bool:
        """删除诊断会话"""
        result = await self.session.execute(
            select(DiagnosisSessionORM).where(
                DiagnosisSessionORM.session_id == session_id
            )
        )
        orm = result.scalar_one_or_none()
        
        if not orm:
            return False
        
        await self.session.delete(orm)
        await self.session.flush()
        return True
    
    async def update(self, diagnosis_session: DiagnosisSession) -> None:
        """更新诊断会话"""
        await self.save(diagnosis_session)
```

```python
import redis.asyncio as redis
from typing import Optional, Any

    职责：
    1. 基础 get/set 操作
    2. 过期时间管理
    3. 连接池管理
    """

        )
    
    async def close(self):
        """关闭连接"""
        if self.redis:
            await self.redis.close()
    
    async def get(self, key: str) -> Optional[str]:
        """获取值"""
        return await self.redis.get(key)
    
    async def set(
        self,
        key: str,
        value: str,
        ex: Optional[int] = None
    ) -> bool:
        """设置值
        
        Args:
            key: 键
            value: 值
            ex: 过期时间（秒）
        """
        return await self.redis.set(key, value, ex=ex)
    
    async def delete(self, key: str) -> bool:

        result = await self.redis.delete(key)
        return result > 0
    
    async def exists(self, key: str) -> bool:

        return await self.redis.exists(key) > 0
    
    async def expire(self, key: str, seconds: int) -> bool:
        """设置过期时间"""
        return await self.redis.expire(key, seconds)
```

```python
"""诊断会话缓存"""
import json
from typing import Optional

from .redis_client import RedisClient
from domain.models.diagnosis_session import DiagnosisSession


class SessionCache:
    """诊断会话缓存
    
    职责：
    1. 缓存正在进行的诊断会话
    2. 减少数据库查询
    """
    
    def __init__(self, redis_client: RedisClient):
        self.redis = redis_client
        self.prefix = "session:"
        self.default_ttl = 3600  # 1 小时
    
    def _make_key(self, session_id: str) -> str:
        """生成缓存键"""
        return f"{self.prefix}{session_id}"
    
    async def set(
        self,
        session_id: str,
        session: DiagnosisSession,
        ttl: Optional[int] = None
    ) -> bool:
        """缓存会话"""
        key = self._make_key(session_id)
        value = json.dumps(session.to_dict())
        ttl = ttl or self.default_ttl
        
        return await self.redis.set(key, value, ex=ttl)
    
    async def get(self, session_id: str) -> Optional[dict]:
        """获取会话"""
        key = self._make_key(session_id)
        value = await self.redis.get(key)
        
        if not value:
            return None
        
        return json.loads(value)
    
    async def delete(self, session_id: str) -> bool:
        """删除会话"""
        key = self._make_key(session_id)
        return await self.redis.delete(key)
    
    async def exists(self, session_id: str) -> bool:
        """检查会话是否存在"""
        key = self._make_key(session_id)
        return await self.redis.exists(key)
```

#### 3.3.3 分布式锁 ⭐ P2

**infrastructure/cache/distributed_lock.py**

```python
"""分布式锁"""
from typing import Optional
from contextlib import asynccontextmanager

from .redis_client import RedisClient


class DistributedLock:
    """Redis 分布式锁
    
    防止重复诊断同一服务
    """
    
    def __init__(self, redis_client: RedisClient):
        self.redis = redis_client
        self.prefix = "lock:"
    
    def _make_key(self, resource: str) -> str:
        """生成锁键"""
        return f"{self.prefix}{resource}"
    
    async def acquire(
        self,
        resource: str,
        ttl: int = 60
    ) -> bool:
        """获取锁
        
        Args:
            resource: 资源标识
            ttl: 锁过期时间（秒）
        
        Returns:
            是否成功获取锁
        """
        key = self._make_key(resource)
        # NX: 仅当键不存在时设置
        result = await self.redis.redis.set(key, "1", ex=ttl, nx=True)
        return result is not None
    
    async def release(self, resource: str) -> bool:
        """释放锁"""
        key = self._make_key(resource)
        return await self.redis.delete(key)
    
    @asynccontextmanager
    async def lock(self, resource: str, ttl: int = 60):
        """锁的上下文管理器
        
        Examples:
            >>> lock = DistributedLock(redis_client)
            >>> async with lock.lock("user-service"):
            ...     # 执行诊断
            ...     pass
        """
        acquired = await self.acquire(resource, ttl)
        if not acquired:
            raise RuntimeError(f"Failed to acquire lock for {resource}")
        
        try:
            yield
        finally:
            await self.release(resource)
```

```python
"""领域事件发布器"""
from typing import Any

from .kafka_producer import KafkaProducer
from domain.events.diagnosis_events import (
    DiagnosisTriggered,
    StageCompleted,
    DiagnosisCompleted,
    DiagnosisFailed
)


class EventPublisher:
    """领域事件发布器
    
    职责：将领域事件发布到 Kafka
    """
    
    def __init__(self, kafka_producer: KafkaProducer):
        self.kafka = kafka_producer
        self.topic = "events.diagnosis"
    
    async def publish(self, event: Any) -> None:
        """发布领域事件"""
        message = self._event_to_message(event)
        await self.kafka.send(
            topic=self.topic,
            message=message,
            key=message.get("diagnosis_id")
        )
    
    def _event_to_message(self, event: Any) -> dict:
        """事件转消息"""
        if isinstance(event, DiagnosisTriggered):
            return {
                "event_type": "DiagnosisTriggered",
                "diagnosis_id": event.diagnosis_id,
                "service_name": event.service_name,
                "triggered_at": event.triggered_at.isoformat()
            }
        elif isinstance(event, StageCompleted):
            return {
                "event_type": "StageCompleted",
                "diagnosis_id": event.diagnosis_id,
                "stage": event.stage.value,
                "progress": event.progress,
                "completed_at": event.completed_at.isoformat()
            }
        elif isinstance(event, DiagnosisCompleted):
            return {
                "event_type": "DiagnosisCompleted",
                "diagnosis_id": event.diagnosis_id,
                "service_name": event.service_name,
                "duration_seconds": event.duration_seconds,
                "completed_at": event.completed_at.isoformat()
            }
        elif isinstance(event, DiagnosisFailed):
            return {
                "event_type": "DiagnosisFailed",
                "diagnosis_id": event.diagnosis_id,
                "service_name": event.service_name,
                "error_message": event.error_message,
                "failed_at": event.failed_at.isoformat()
            }
        else:
            raise ValueError(f"Unknown event type: {type(event)}")
```
