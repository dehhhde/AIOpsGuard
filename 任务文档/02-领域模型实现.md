```python
"""共享值对象"""
from dataclasses import dataclass
from datetime import datetime
from typing import Optional


@dataclass(frozen=True)
class TimeRange:
    """
    时间范围值对象
    
    职责：封装时间范围的业务逻辑
    不可变：使用 frozen=True
    """
    start_time: datetime
    end_time: datetime
    
    def __post_init__(self):
        """验证时间范围"""
        if self.start_time >= self.end_time:
            raise ValueError(
                f"start_time ({self.start_time}) must be before end_time ({self.end_time})"
            )
    
    @property
    def duration_seconds(self) -> int:
        """持续时间（秒）"""
        return int((self.end_time - self.start_time).total_seconds())
    
    @property
    def duration_minutes(self) -> float:
        """持续时间（分钟）"""
        return self.duration_seconds / 60
    
    def contains(self, timestamp: datetime) -> bool:
        """判断时间点是否在范围内"""
        return self.start_time <= timestamp <= self.end_time
    
    def overlaps(self, other: 'TimeRange') -> bool:
        """判断是否与另一个时间范围重叠"""
        return (self.start_time <= other.end_time and 
                self.end_time >= other.start_time)
    
    def to_dict(self) -> dict:
        """转换为字典"""
        return {
            "start_time": self.start_time.isoformat(),
            "end_time": self.end_time.isoformat(),
            "duration_seconds": self.duration_seconds
        }
    
    @classmethod
    def from_minutes_ago(cls, minutes: int) -> 'TimeRange':
        """从 N 分钟前到现在"""
        from datetime import timedelta
        end_time = datetime.now()
        start_time = end_time - timedelta(minutes=minutes)
        return cls(start_time, end_time)
    
    @classmethod
    def from_hours_ago(cls, hours: int) -> 'TimeRange':
        """从 N 小时前到现在"""
        from datetime import timedelta
        end_time = datetime.now()
        start_time = end_time - timedelta(hours=hours)
        return cls(start_time, end_time)


@dataclass(frozen=True)
class MetricPoint:
    """
    指标数据点值对象
    
    职责：表示单个时间点的指标值
    """
    timestamp: datetime
    value: float
    labels: tuple = ()  # 使用 tuple 保证不可变
    
    def __post_init__(self):
        """验证数据"""
        if not isinstance(self.labels, tuple):
            # 转换为 tuple
            object.__setattr__(self, 'labels', tuple(self.labels))
    
    def to_dict(self) -> dict:
        """转换为字典"""
        return {
            "timestamp": self.timestamp.isoformat(),
            "value": self.value,
            "labels": dict(self.labels) if self.labels else {}
        }
```

```python
"""基础实体类"""
from abc import ABC
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional
import uuid


@dataclass
class Entity(ABC):
    """
    实体基类
    
    特征：
    1. 有唯一标识（ID）
    2. 可变
    3. 通过 ID 判断相等性
    """
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)
    
    def __eq__(self, other) -> bool:
        """通过 ID 判断相等"""
        if not isinstance(other, self.__class__):
            return False
        return self.id == other.id
    
    def __hash__(self) -> int:
        """支持作为字典键和集合元素"""
        return hash(self.id)
    
    def _update_timestamp(self) -> None:
        """更新时间戳"""
        self.updated_at = datetime.now()


@dataclass
class AggregateRoot(Entity):
    """
    聚合根基类
    
    职责：
    1. 维护聚合内的一致性边界
    2. 管理领域事件
    3. 作为外部访问聚合的唯一入口
    """
    _domain_events: list = field(default_factory=list, init=False, repr=False)
    
    def add_domain_event(self, event) -> None:
        """添加领域事件"""
        self._domain_events.append(event)
    
    def clear_domain_events(self) -> list:
        """清空并返回领域事件"""
        events = self._domain_events.copy()
        self._domain_events.clear()
        return events
    
    @property
    def domain_events(self) -> list:
        """获取领域事件（只读）"""
        return self._domain_events.copy()
```

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional

from shared.kernel.base_entity import Entity
from shared.constants.enums import EvidenceType


@dataclass
class Evidence(Entity):
    """
    证据实体
    
    职责：记录诊断过程中收集的证据
    生命周期：从创建到诊断完成
    """
    type: EvidenceType
    source: str                          # 证据来源（如 metric name, log pattern）
    description: str                     # 证据描述
    detail: Optional[str] = None         # 详细信息（如堆栈、链路 ID）
    weight: float = 1.0                  # 证据权重（0-1）
    timestamp: datetime = field(default_factory=datetime.now)
    metadata: dict = field(default_factory=dict)  # 额外元数据
    
    def __post_init__(self):
        """验证数据"""
        super().__post_init__()
        
        if not 0 <= self.weight <= 1:
            raise ValueError(f"Weight must be between 0 and 1, got {self.weight}")
        
        if not self.description:
            raise ValueError("Description cannot be empty")
        
        if not self.source:
            raise ValueError("Source cannot be empty")
    
    def is_strong(self) -> bool:
        """是否强证据（权重 >= 0.7）"""
        return self.weight >= 0.7
    
    def is_weak(self) -> bool:
        """是否弱证据（权重 < 0.3）"""
        return self.weight < 0.3
    
    def add_metadata(self, key: str, value: any) -> None:
        """添加元数据"""
        self.metadata[key] = value
        self._update_timestamp()
    
    def to_dict(self) -> dict:
        """转换为字典"""
        return {
            "id": self.id,
            "type": self.type.value,
            "source": self.source,
            "description": self.description,
            "detail": self.detail,
            "weight": self.weight,
            "timestamp": self.timestamp.isoformat(),
```

```python
@dataclass(frozen=True)
class RootCause:
    """
    根因值对象
    
    职责：封装根因分析结果
    不可变：诊断完成后不应修改
    """
    problem_type: ProblemType
    description: str
    confidence: float

            raise ValueError(

            )
        
        if not self.description:
            raise ValueError("Description cannot be empty")
        
        if not self.reasoning:
            raise ValueError("Reasoning cannot be empty")
        
        # 确保 recommendations 是 tuple
        if not isinstance(self.recommendations, tuple):
            object.__setattr__(
                self, 'recommendations', tuple(self.recommendations)
            )
    
    def is_high_confidence(self) -> bool:
        """是否高置信度（>= 0.8）"""
        return self.confidence >= 0.8
    
    def is_medium_confidence(self) -> bool:
        """是否中等置信度（0.5 ~ 0.8）"""
        return 0.5 <= self.confidence < 0.8
    
    def is_low_confidence(self) -> bool:
        """是否低置信度（< 0.5）"""
        return self.confidence < 0.5
    
    def to_dict(self) -> dict:
        """转换为字典"""
        return {
            "problem_type": self.problem_type.value,
            "description": self.description,
            "confidence": self.confidence,
            "reasoning": self.reasoning,
            "recommendations": list(self.recommendations)
```

```python
    """
    指标异常实体
    
    职责：表示单个指标的异常情况

    current_value: float = 0.0
    baseline: float = 0.0
    deviation: float = 0.0  # 偏离度（0-1）
    
    def __post_init__(self):
        """自动计算严重程度"""
        super().__post_init__()

            object.__setattr__(self, 'severity', self._calculate_severity())

        """根据偏离度计算严重程度"""
        if self.deviation >= 0.8:
            return SeverityLevel.CRITICAL
        elif self.deviation >= 0.5:
            return SeverityLevel.HIGH
        elif self.deviation >= 0.3:
            return SeverityLevel.MEDIUM
        else:
            return SeverityLevel.LOW
    
    def to_dict(self) -> dict:
        """转换为字典"""
        base_dict = super().to_dict()
        base_dict.update({
            "metric_name": self.metric_name,
            "current_value": self.current_value,
            "baseline": self.baseline,
            "deviation": self.deviation
```

**domain/models/diagnosis.py**

```python
"""诊断会话聚合根"""
from dataclasses import dataclass, field
from datetime import datetime
from typing import List, Optional

from shared.kernel.base_entity import AggregateRoot
from shared.kernel.value_objects import TimeRange
from shared.constants.enums import DiagnosisStatus
from domain.models.evidence import Evidence
from domain.models.value_objects import RootCause
from domain.events.diagnosis_events import (
    DiagnosisCreated,
    DiagnosisStatusChanged,
    EvidenceAdded,
    RootCauseIdentified,
    DiagnosisCompleted
)


@dataclass
class DiagnosisSession(AggregateRoot):
    """
    诊断会话聚合根
    
    职责：
    1. 管理诊断生命周期
    2. 维护诊断状态一致性
    3. 协调证据和根因
    4. 发布领域事件
    
    聚合边界：
    - Evidence 实体（聚合内部）
    - RootCause 值对象（聚合内部）
    """
    diagnosis_id: str
    target_service: str
    time_range: TimeRange
    status: DiagnosisStatus = DiagnosisStatus.PENDING
    evidences: List[Evidence] = field(default_factory=list)
    root_causes: List[RootCause] = field(default_factory=list)
    error_message: Optional[str] = None
    metadata: dict = field(default_factory=dict)
    
    def __post_init__(self):
        """初始化后发布创建事件"""
        super().__post_init__()
        
        # 发布诊断创建事件
        self.add_domain_event(
            DiagnosisCreated(
                diagnosis_id=self.diagnosis_id,
                target_service=self.target_service,
                time_range=self.time_range,
                created_at=self.created_at
            )
        )
    
    def add_evidence(self, evidence: Evidence) -> None:
        """
        添加证据
        
        业务规则：
        1. 只有在 COLLECTING 或 ANALYZING 状态才能添加证据
        2. 不能添加重复的证据
        """
        # 验证状态
        if self.status in [DiagnosisStatus.COMPLETED, DiagnosisStatus.FAILED]:
            raise ValueError(
                f"Cannot add evidence in status {self.status.value}"
            )
        
        # 验证重复
        if any(e.id == evidence.id for e in self.evidences):
            raise ValueError(f"Evidence {evidence.id} already exists")
        
        self.evidences.append(evidence)
        self._update_timestamp()
        
        # 发布证据添加事件
        self.add_domain_event(
            EvidenceAdded(
                diagnosis_id=self.diagnosis_id,
                evidence=evidence,
                total_evidences=len(self.evidences)
            )
        )
    
    def add_evidences(self, evidences: List[Evidence]) -> None:
        """批量添加证据"""
        for evidence in evidences:
            self.add_evidence(evidence)
    
    def add_root_cause(self, root_cause: RootCause) -> None:
        """
        添加根因
        
        业务规则：
        1. 只有在 ANALYZING 状态才能添加根因
        2. 自动按置信度排序
        """
        if self.status != DiagnosisStatus.ANALYZING:
            raise ValueError(
                f"Cannot add root cause in status {self.status.value}"
            )
        
        self.root_causes.append(root_cause)
        # 按置信度降序排序
        self.root_causes.sort(key=lambda rc: rc.confidence, reverse=True)
        self._update_timestamp()
        
        # 发布根因识别事件
        self.add_domain_event(
            RootCauseIdentified(
                diagnosis_id=self.diagnosis_id,
                root_cause=root_cause,
                total_root_causes=len(self.root_causes)
            )
        )
    
    def transition_to(self, new_status: DiagnosisStatus) -> None:
        """
        状态转换
        
        业务规则：状态机验证
        PENDING -> COLLECTING -> ANALYZING -> COMPLETED
                                         \-> FAILED
        """
        # 定义有效的状态转换
        valid_transitions = {
            DiagnosisStatus.PENDING: [DiagnosisStatus.COLLECTING],
            DiagnosisStatus.COLLECTING: [
                DiagnosisStatus.ANALYZING,
                DiagnosisStatus.FAILED,
                DiagnosisStatus.TIMEOUT
            ],
            DiagnosisStatus.ANALYZING: [
                DiagnosisStatus.COMPLETED,
                DiagnosisStatus.FAILED,
                DiagnosisStatus.TIMEOUT
            ],
            DiagnosisStatus.COMPLETED: [],
            DiagnosisStatus.FAILED: [],
            DiagnosisStatus.TIMEOUT: []
        }
        
        # 验证转换
        if new_status not in valid_transitions[self.status]:
            raise ValueError(
                f"Invalid transition from {self.status.value} to {new_status.value}"
            )
        
        old_status = self.status
        self.status = new_status
        self._update_timestamp()
        
        # 发布状态变更事件
        self.add_domain_event(
            DiagnosisStatusChanged(
                diagnosis_id=self.diagnosis_id,
                old_status=old_status,
                new_status=new_status,
                changed_at=datetime.now()
            )
        )
        
        # 如果完成，发布完成事件
        if new_status == DiagnosisStatus.COMPLETED:
            self.add_domain_event(
                DiagnosisCompleted(
                    diagnosis_id=self.diagnosis_id,
                    target_service=self.target_service,
                    primary_root_cause=self.get_primary_root_cause(),
                    confidence=self.calculate_overall_confidence(),
                    evidence_count=len(self.evidences),
                    completed_at=datetime.now()
                )
            )
    
    def mark_as_failed(self, error_message: str) -> None:
        """标记为失败"""
        self.error_message = error_message
        self.transition_to(DiagnosisStatus.FAILED)
    
    def get_primary_root_cause(self) -> Optional[RootCause]:
        """获取主根因（置信度最高）"""
        return self.root_causes[0] if self.root_causes else None
    
    def calculate_overall_confidence(self) -> float:
        """计算整体置信度"""
        if not self.root_causes:
            return 0.0
        
        # 取最高置信度
        return self.root_causes[0].confidence
    
    def get_strong_evidences(self) -> List[Evidence]:
        """获取强证据（权重 >= 0.7）"""
        return [e for e in self.evidences if e.is_strong()]
    
    def get_evidences_by_type(self, evidence_type) -> List[Evidence]:
        """按类型获取证据"""
        return [e for e in self.evidences if e.type == evidence_type]
    
    def is_in_progress(self) -> bool:
        """是否进行中"""
        return self.status in [
            DiagnosisStatus.COLLECTING,
            DiagnosisStatus.ANALYZING
        ]
    
    def is_finished(self) -> bool:
        """是否已结束"""
        return self.status in [
            DiagnosisStatus.COMPLETED,
            DiagnosisStatus.FAILED,
            DiagnosisStatus.TIMEOUT
        ]
    
    def add_metadata(self, key: str, value: any) -> None:
        """添加元数据"""
        self.metadata[key] = value
        self._update_timestamp()
    
    def to_dict(self) -> dict:
        """转换为字典"""
        return {
            "id": self.id,
            "diagnosis_id": self.diagnosis_id,
            "target_service": self.target_service,
            "time_range": self.time_range.to_dict(),
            "status": self.status.value,
            "evidences": [e.to_dict() for e in self.evidences],
            "root_causes": [rc.to_dict() for rc in self.root_causes],
            "primary_root_cause": (
                self.get_primary_root_cause().to_dict()
                if self.get_primary_root_cause() else None
            ),
            "overall_confidence": self.calculate_overall_confidence(),
            "error_message": self.error_message,
            "metadata": self.metadata,
            "created_at": self.created_at.isoformat(),
            "updated_at": self.updated_at.isoformat()
        }
```

```python
from dataclasses import dataclass
from datetime import datetime
from typing import Optional

from shared.kernel.value_objects import TimeRange
from shared.constants.enums import DiagnosisStatus


@dataclass
class DomainEvent:
    """领域事件基类"""
    event_id: str
    occurred_at: datetime
    
    def __post_init__(self):
        """自动生成事件 ID 和时间"""
        import uuid
        if not self.event_id:
            object.__setattr__(self, 'event_id', str(uuid.uuid4()))
        if not self.occurred_at:
            object.__setattr__(self, 'occurred_at', datetime.now())


@dataclass
class DiagnosisCreated(DomainEvent):

    diagnosis_id: str
    target_service: str
    time_range: TimeRange

    event_id: str = ""
```

## 🚨 常见问题

### 问题1：frozen dataclass 无法修改属性

**症状**：尝试修改值对象属性时报错

**解决方案**：
```python
# 错误做法
root_cause.confidence = 0.8  # FrozenInstanceError

# 正确做法：创建新对象
root_cause = RootCause(
    problem_type=ProblemType.MEMORY_LEAK,
    description="...",
    confidence=0.8,  # 新值
    reasoning="...",
    recommendations=(...)
)
```

### 问题2：循环导入

**症状**：`ImportError: cannot import name 'X' from partially initialized module`

**解决方案**：
```python
# 使用类型注解的字符串形式
from typing import Optional

class DiagnosisSession:
    root_causes: List['RootCause']  # 字符串形式

# 或使用 TYPE_CHECKING
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from domain.models.value_objects import RootCause
```
