#### 1.2.5 创建主应用入口

**main.py**

```python
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from contextlib import asynccontextmanager
import structlog

from config.settings import settings
from adapters.rest.routers import diagnosis_router, health_router
from infrastructure.persistence.database import init_db, close_db
from infrastructure.monitoring.metrics import setup_metrics

logger = structlog.get_logger()


@asynccontextmanager
async def lifespan(app: FastAPI):
    """应用生命周期管理"""
    # 启动时执行
    logger.info("Starting AIOps Diagnosis Agent", version=settings.app_version)
    
    # 初始化数据库
    await init_db()
    
    # 设置监控指标
    setup_metrics()
    
    logger.info("Application started successfully")
    
    yield
    
    # 关闭时执行
    logger.info("Shutting down AIOps Diagnosis Agent")
    await close_db()
    logger.info("Application shutdown complete")


def create_app() -> FastAPI:
    """创建 FastAPI 应用"""
    
    app = FastAPI(
        title=settings.app_name,
        version=settings.app_version,
        debug=settings.debug,
        lifespan=lifespan
    )
    
    # 配置 CORS
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"],  # 生产环境需要限制
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )
    
    # 注册路由
    app.include_router(health_router.router, tags=["Health"])
    app.include_router(
        diagnosis_router.router,
        prefix="/api/v1/diagnosis",
        tags=["Diagnosis"]
    )
    
    return app


app = create_app()


if __name__ == "__main__":
    import uvicorn
    
    uvicorn.run(
        "main:app",
        host="0.0.0.0",
        port=8000,
        reload=settings.debug,
        log_level=settings.log_level.lower()
    )
```

```powershell
# 健康检查脚本
Write-Host "Checking infrastructure health..." -ForegroundColor Green

# 检查 PostgreSQL
try {
    $pgResult = docker exec aiops-postgres pg_isready -U postgres
    Write-Host "✓ PostgreSQL: OK" -ForegroundColor Green
} catch {
    Write-Host "✗ PostgreSQL: FAILED" -ForegroundColor Red
}

# 检查 Redis
try {
    $redisResult = docker exec aiops-redis redis-cli ping
    if ($redisResult -eq "PONG") {
        Write-Host "✓ Redis: OK" -ForegroundColor Green
    }
} catch {
    Write-Host "✗ Redis: FAILED" -ForegroundColor Red
}

# 检查 Prometheus
try {
    $promResult = Invoke-WebRequest -Uri "http://localhost:9090/-/healthy" -UseBasicParsing
    if ($promResult.StatusCode -eq 200) {

}
```

## 🚨 常见问题

### 问题1：Docker 启动失败

**症状**：`docker-compose up` 报错

**解决方案**：
```powershell
# 检查 Docker Desktop 是否运行
docker info

# 检查端口占用
netstat -ano | findstr "5432"
netstat -ano | findstr "6379"
netstat -ano | findstr "9090"

# 如有端口冲突，修改 docker-compose.yml 中的端口映射
```

### 问题2：Poetry 安装依赖慢

**解决方案**：
```powershell
# 配置国内镜像源
poetry config repositories.aliyun https://mirrors.aliyun.com/pypi/simple/
poetry config pypi-mirror.url https://mirrors.aliyun.com/pypi/simple/
```

### 问题3：Python 虚拟环境激活失败

**解决方案**：
```powershell
# Windows PowerShell 执行策略限制
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser

# 然后重新激活
.\venv\Scripts\activate
```
